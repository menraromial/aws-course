\section{Projet final}

\begin{frame}{Objectif}
  \begin{itemize}
    \setlength{\itemsep}{.5em}
    \item Déployer une application web publique sur une instance EC2
    \item Fonctions : déposer un fichier (10 Mo au plus), lister les fichiers, les télécharger
    \item Fichiers stockés dans un bucket S3 privé
    \item Application fournie : Flask + boto3 ; travail demandé : le déploiement
  \end{itemize}
  \bigskip
  \textbf{Exigences}
  \begin{itemize}
    \item HTTP (80) et HTTPS (443) ouverts à tous, SSH (22) limité à votre adresse
    \item Aucune clé d'accès sur l'instance : accès à S3 par un rôle IAM
    \item Rôle limité au bucket : lister, lire, écrire, pas de suppression
    \item Redémarrage automatique de l'application (panne, redémarrage de l'instance)
  \end{itemize}
  \medskip
  \textbf{Partie 2} : deux instances dans deux AZ, derrière un Application Load Balancer
\end{frame}

\begin{frame}{URL présignées}
  \begin{itemize}
    \setlength{\itemsep}{.55em}
    \item URL signée avec les identifiants de l'application, pour un objet et une durée donnés
    \item Générée localement, sans appel réseau
    \item Le navigateur télécharge directement depuis S3 : pas de charge sur l'instance
    \item Durée de validité limitée par celle des identifiants : quelques heures avec un rôle, 7 jours au maximum avec une clé permanente
  \end{itemize}
  \lien{https://docs.aws.amazon.com/AmazonS3/latest/userguide/using-presigned-url.html}
\end{frame}

\begin{frame}{HTTPS}
  \begin{itemize}
    \setlength{\itemsep}{.55em}
    \item Certificat reconnu par les navigateurs : délivré pour un nom de domaine
    \item Projet : certificat autosigné, connexion chiffrée mais avertissement du navigateur
    \item En production :
    \begin{itemize}
      \item nom de domaine (Route 53 ou autre registraire)
      \item certificat gratuit AWS Certificate Manager, sur un répartiteur de charge ou CloudFront
      \item ou certificat Let's Encrypt directement sur Nginx
    \end{itemize}
  \end{itemize}
\end{frame}

    Moindre privilège & \texttt{aws s3 rm} refusé \\
    Résilience & relance après \texttt{kill -9} et après redémarrage \\ \bottomrule
  \end{tabularx}

\begin{frame}{Répartition de charge : Elastic Load Balancing}
  \begin{itemize}
    \setlength{\itemsep}{.55em}
    \item Distribue le trafic entrant entre plusieurs cibles : instances EC2, conteneurs, adresses IP, dans une ou plusieurs AZ
    \item Contrôle de santé des cibles : trafic envoyé aux seules cibles en bonne santé
    \item Capacité du répartiteur ajustée automatiquement au trafic
    \item Un seul point d'entrée (nom DNS) devant des serveurs interchangeables
    \item Condition : application sans état local (fichiers, sessions dans un service partagé : S3, base, cache)
  \end{itemize}
  \lien{https://docs.aws.amazon.com/elasticloadbalancing/latest/userguide/what-is-load-balancing.html}
\end{frame}

\begin{frame}{Les types de répartiteurs}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{3.5cm}>{\raggedright\arraybackslash}p{2.3cm}>{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Type} & \textbf{Couche OSI} & \textbf{Usage} \\ \midrule
    \textbf{Application LB} (ALB) & 7, application & HTTP, HTTPS, gRPC ; routage par chemin, nom d'hôte, en-têtes ; sites et API web \\
    \textbf{Network LB} (NLB) & 4, transport & TCP, UDP, TLS ; millions de requêtes par seconde ; adresse IP fixe par AZ \\
    \textbf{Gateway LB} (GWLB) & 3, réseau & tous les paquets IP ; appliances virtuelles (pare-feu, détection d'intrusion), protocole GENEVE, port 6081 \\
    \textbf{Classic LB} (CLB) & 4 et 7 & génération précédente ; migration recommandée \\ \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}{Application Load Balancer}
  \begin{itemize}
    \setlength{\itemsep}{.45em}
    \item \textbf{Écouteur} (\textit{listener}) : port et protocole (HTTP:80), règles vers un groupe cible
    \item \textbf{Groupe cible} (\textit{target group}) : les instances et leur port (80)
    \item \textbf{Contrôle de santé} : toutes les 30 s par défaut ; 2 échecs pour écarter une cible, 5 succès pour la réintégrer
    \item Au moins deux AZ ; répartition entre AZ toujours active
    \item Security Group des instances : HTTP depuis le Security Group du répartiteur
    \item Prix : à l'heure, plus unités de capacité (LCU) ; environ 0,03 \$ par heure pour un petit site
  \end{itemize}
  \lien{https://docs.aws.amazon.com/elasticloadbalancing/latest/application/introduction.html}
\end{frame}

\begin{frame}{Elastic Beanstalk}
  \begin{itemize}
    \setlength{\itemsep}{.5em}
    \item Service géré (PaaS) : une archive zip et une plateforme (Python 3.12, AL2023)
    \item Crée instances, groupe Auto Scaling, répartiteur, Security Groups ; installe Nginx et Gunicorn
    \item Chaque environnement est une pile CloudFormation ; versions rangées dans \texttt{elasticbeanstalk-<région>-<compte>}
    \item Instance unique (Elastic IP) ou répartiteur (ALB) ; service gratuit, on paie les ressources
    \item Profil d'instance obligatoire : stratégie \texttt{AWSElasticBeanstalkWebTier}
    \item Comptes créés après le 1/10/2024 : modèles de lancement obligatoires (IMDSv1 désactivé ou volume gp3)
  \end{itemize}
  \lien{https://docs.aws.amazon.com/elasticbeanstalk/latest/dg/Welcome.html}
\end{frame}

\begin{frame}{Politiques de déploiement}
  \small\renewcommand{\arraystretch}{1.2}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{4.4cm}>{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Politique} & \textbf{Fonctionnement} \\ \midrule
    All at once (défaut) & toutes les instances en même temps ; brève interruption \\
    Rolling & par lots ; chaque lot sort du répartiteur ; capacité réduite \\
    Rolling with additional batch & lot supplémentaire lancé d'abord ; capacité maintenue \\
    Immutable & nouvelles instances dans un groupe Auto Scaling à part \\
    Traffic splitting & comme Immutable, une part du trafic d'abord vers la nouvelle version \\ \bottomrule
  \end{tabularx}
